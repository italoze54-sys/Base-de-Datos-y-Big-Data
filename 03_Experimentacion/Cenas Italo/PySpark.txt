from pyspark.sql import functions as F

ruta = "/Volumes/workspace/datostottusbd/sedetottus"

df_sede = spark.read.option("header", True).option("inferSchema", True).csv(f"{ruta}/Sede_Tottus.csv")
df_categoria = spark.read.option("header", True).option("inferSchema", True).csv(f"{ruta}/Categoria.csv")
df_producto = spark.read.option("header", True).option("inferSchema", True).csv(f"{ruta}/Producto.csv")
df_beneficiario = spark.read.option("header", True).option("inferSchema", True).csv(f"{ruta}/Beneficiario.csv")

print(f"Catálogos leídos -> Sedes: {df_sede.count()}, Categorías: {df_categoria.count()}, Productos: {df_producto.count()}, Beneficiarios: {df_beneficiario.count()}")

spark.sql("USE CATALOG workspace")
spark.sql("USE SCHEMA datostottusbd")

# CAMBIAR ESTE VALOR PARA CADA ESCALA: 1000 -> 10000 -> 100000 -> 1000000
N_FILAS = 1000000

# 1. Lote_merma
df_lote = spark.range(1, N_FILAS + 1).withColumnRenamed("id", "id_lote") \
    .withColumn("id_producto", ((F.col("id_lote") - 1) % 100) + 1) \
    .withColumn("id_sede", ((F.col("id_lote") - 1) % 20) + 1) \
    .withColumn("fecha_ingreso_merma", F.date_add(F.lit("2026-06-01"), (F.col("id_lote") % 90).cast("int"))) \
    .withColumn("fecha_vencimiento", F.date_add(F.col("fecha_ingreso_merma"), 3)) \
    .withColumn("cantidad_disponible", F.round(F.lit(10.00) + (F.col("id_lote") % 40), 2)) \
    .withColumn("tipo_destino", F.when(F.col("id_lote") % 2 == 0, "Donación Social").otherwise("Venta Preferencial")) \
    .withColumn("precio_rescate", F.round(F.lit(1.50) + (F.col("id_lote") % 15) * 0.50, 2)) \
    .withColumn("estado_lote", F.when(F.col("id_lote") % 3 == 0, "Agotado").otherwise("Disponible"))

# 2. Pedido_rescate
df_pedido = df_lote.select(
    F.col("id_lote").alias("id_pedido"),
    (((F.col("id_lote") - 1) % 100) + 1).alias("id_beneficiario"),
    F.col("id_sede"),
    F.date_add(F.col("fecha_ingreso_merma"), 1).alias("fecha_solicitud"),
    F.round(F.col("precio_rescate") * 5.00, 2).alias("costo_total"),
    F.when(F.col("id_lote") % 6 == 0, "Pendiente").otherwise("Entregado").alias("estado_pedido")
)

# 3. detalle_pedido_rescate
df_detalle = df_pedido.select(
    F.col("id_pedido").alias("id_detalle"),
    F.col("id_pedido"),
    F.col("id_pedido").alias("id_lote"),
    F.lit(5.00).alias("cantidad_solicitada"),
    F.col("costo_total").alias("subtotal")
)

# 4. Entrega_despacho
df_despacho = df_pedido.select(
    F.col("id_pedido").alias("id_entrega"),
    F.col("id_pedido"),
    F.when(F.col("estado_pedido") == "Pendiente", "Pendiente").otherwise("Conforme").alias("conformidad_recibida")
)

# Guardar en tablas Delta dentro de workspace.datostottusbd
df_lote.write.format("delta").mode("overwrite").saveAsTable("oltp_lote_merma")
df_pedido.write.format("delta").mode("overwrite").saveAsTable("oltp_pedido_rescate")
df_detalle.write.format("delta").mode("overwrite").saveAsTable("oltp_detalle_pedido")
df_despacho.write.format("delta").mode("overwrite").saveAsTable("oltp_entrega_despacho")

print(f"Tablas transaccionales listas en workspace.datostottusbd con {N_FILAS:,} filas cada una.")

import time

tiempos_ms = []

for intento in range(1, 4):
    t0 = time.perf_counter()

    # Extracción
    pr = spark.table("oltp_pedido_rescate")
    dpr = spark.table("oltp_detalle_pedido")
    lm = spark.table("oltp_lote_merma")

    # Transformación y Cruce Dimensional (idéntico a sp_Ejecutar_ETL_Tottus)
    df_fact = pr.join(dpr, "id_pedido") \
        .join(lm, dpr.id_lote == lm.id_lote) \
        .join(df_producto, lm.id_producto == df_producto.id_producto) \
        .join(df_beneficiario, pr.id_beneficiario == df_beneficiario.id_beneficiario) \
        .join(df_sede, pr.id_sede == df_sede.id_sede) \
        .withColumn("id_tiempo", F.date_format(pr.fecha_solicitud, "yyyyMMdd").cast("int")) \
        .withColumn("id_canal",
            F.when(df_beneficiario.tipo_beneficiario == "Olla Comun", 1)
             .when(df_beneficiario.tipo_beneficiario == "Comedor Popular", 2)
             .when(df_beneficiario.tipo_beneficiario == "Bodega Aliada MYPE", 3)
             .otherwise(4)
        ) \
        .groupBy(
            "id_tiempo",
            df_producto.id_producto,
            lm.id_lote,
            df_sede.id_sede,
            df_beneficiario.id_beneficiario,
            "id_canal"
        ).agg(
            F.sum(dpr.cantidad_solicitada).alias("cantidad_rescatada"),
            F.lit(0.00).alias("cantidad_merma_perdida"),
            F.lit(0.00).alias("monto_perdida_merma"),
            F.round(F.sum(dpr.subtotal), 2).alias("monto_recuperado_rescate"),
            F.round(F.sum(dpr.cantidad_solicitada * df_producto.precio_original) - F.sum(dpr.subtotal), 2).alias("ahorro_social_donado")
        )

    # Carga en Data Mart Delta
    df_fact.write.format("delta").mode("overwrite").saveAsTable("dm_fact_gestion_merma_rescate")

    t1 = time.perf_counter()
    ms = (t1 - t0) * 1000
    tiempos_ms.append(ms)
    print(f"Repetición {intento}: {ms:.2f} ms ({ms/1000:.3f} s)")

promedio_ms = sum(tiempos_ms) / len(tiempos_ms)
throughput = N_FILAS / (promedio_ms / 1000.0)
cuadre = spark.table("dm_fact_gestion_merma_rescate").agg(
    F.count("*").alias("Total_Filas"),
    F.sum("cantidad_rescatada").alias("Total_Unidades"),
    F.round(F.sum("monto_recuperado_rescate"), 2).alias("Total_Soles")
).collect()[0]

print("-" * 60)
print(f"ESCALA EVALUADA: {N_FILAS:,} registros")
print(f"Tiempo Promedio: {promedio_ms:.2f} ms ({promedio_ms/1000:.3f} s)")
print(f"Throughput:      {throughput:,.2f} filas/segundo")
print(f"Reconciliación:  {cuadre['Total_Filas']:,} filas | {cuadre['Total_Unidades']:,.2f} unid | S/ {cuadre['Total_Soles']:,.2f}")
